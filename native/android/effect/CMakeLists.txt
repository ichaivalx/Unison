cmake_minimum_required(VERSION 3.22)
project(unison_tap LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AIDL_SRC "${CMAKE_CURRENT_SOURCE_DIR}/build/aidl/src")
set(PLATFORM_CPP "${CMAKE_CURRENT_SOURCE_DIR}/build/platform-cxx")
add_library(platform_cpp SHARED IMPORTED)
set_target_properties(platform_cpp PROPERTIES IMPORTED_LOCATION "${PLATFORM_CPP}/lib/libc++.so")
add_library(platform_binder SHARED build/binder_stub.c)
set_target_properties(platform_binder PROPERTIES OUTPUT_NAME binder_ndk)
add_library(platform_cutils SHARED build/cutils_stub.c)
set_target_properties(platform_cutils PROPERTIES OUTPUT_NAME cutils)
add_library(platform_abi INTERFACE)
target_include_directories(platform_abi INTERFACE build/aidl/include)
target_compile_options(platform_abi INTERFACE -nostdinc++
    "SHELL:-isystem ${PLATFORM_CPP}/config" "SHELL:-isystem ${PLATFORM_CPP}/include")
target_compile_definitions(platform_abi INTERFACE __ANDROID_VENDOR__=1 BINDER_STABILITY_SUPPORT=1)
target_link_options(platform_abi INTERFACE -nostdlib++)
target_link_libraries(platform_abi INTERFACE platform_cpp platform_binder)
# CI 生成的同名共享库仅用于链接。模块不携带它们，运行时复用设备的绑定与 Binder class。
function(binding target soname directory)
    file(GLOB sources CONFIGURE_DEPENDS "${AIDL_SRC}/${directory}/*.cpp")
    add_library(${target} SHARED ${sources})
    set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "${soname}")
    target_include_directories(${target} PUBLIC build/aidl/include)
    target_compile_options(${target} PRIVATE -fno-rtti)
    target_link_libraries(${target} PUBLIC platform_abi)
endfunction()
binding(hardware_common android.hardware.common-V2-ndk android/hardware/common)
binding(fmq_binding android.hardware.common.fmq-V1-ndk android/hardware/common/fmq)
binding(audio_types android.media.audio.common.types-V4-ndk android/media/audio/common)
binding(audio_common android.hardware.audio.common-V3-ndk android/hardware/audio/common)
binding(effect_binding android.hardware.audio.effect-V2-ndk android/hardware/audio/effect)
target_link_libraries(fmq_binding PUBLIC hardware_common)
target_link_libraries(audio_common PUBLIC audio_types)
target_link_libraries(effect_binding PUBLIC audio_common audio_types hardware_common fmq_binding)
add_library(unison_tap SHARED unison_effect.cpp)
set_target_properties(unison_tap PROPERTIES SKIP_BUILD_RPATH TRUE)
target_include_directories(unison_tap PRIVATE build/aidl/include)
target_compile_options(unison_tap PRIVATE -Wall -Wextra -fno-rtti -fvisibility=hidden -fvisibility-inlines-hidden
    -Xclang -fdump-vtable-layouts)
target_link_libraries(unison_tap PRIVATE effect_binding platform_abi platform_cutils log)
target_link_options(unison_tap PRIVATE -Wl,--no-undefined -Wl,--exclude-libs,ALL -Wl,--as-needed)
add_executable(unison_hal_watch hal_watch.c)
target_include_directories(unison_hal_watch PRIVATE build/aidl/include)
target_link_libraries(unison_hal_watch PRIVATE platform_binder)
