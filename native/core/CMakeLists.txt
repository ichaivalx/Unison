cmake_minimum_required(VERSION 3.22)
project(unison_core LANGUAGES CXX)
add_library(unison-core STATIC receiver.cpp)
set_target_properties(unison-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(unison-core PUBLIC cxx_std_17)
target_include_directories(unison-core PUBLIC include)
if(WIN32)
    target_compile_definitions(unison-core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(unison-core PUBLIC ws2_32)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(unison-core PUBLIC Threads::Threads)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    enable_testing()
    add_executable(receiver-test receiver_test.cpp)
    target_link_libraries(receiver-test PRIVATE unison-core)
    add_test(NAME receiver-identity COMMAND receiver-test)
endif()
