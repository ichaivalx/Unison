cmake_minimum_required(VERSION 3.22)
project(unison_android LANGUAGES C CXX)
get_filename_component(PLUGIN_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
get_filename_component(UNISON_ROOT "${PLUGIN_CPP_DIR}/../../../../../.." ABSOLUTE)
add_subdirectory("${UNISON_ROOT}/native/core" "${CMAKE_CURRENT_BINARY_DIR}/core")
add_library(unison SHARED receiver_android.cpp)
target_compile_features(unison PRIVATE cxx_std_17)
target_link_libraries(unison PRIVATE unison-core aaudio)
add_library(unison_capture SHARED capture_signals.c effect_socket.cpp calibration_match.cpp)
target_include_directories(unison_capture PRIVATE "${UNISON_ROOT}/native/android/effect")
target_compile_features(unison_capture PRIVATE cxx_std_17)
target_compile_options(unison_capture PRIVATE -O2 -Wall -Wextra -Werror)
