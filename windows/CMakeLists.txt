cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(unison_windows LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(../core "${CMAKE_CURRENT_BINARY_DIR}/core")
add_executable(unison-receiver main.cpp)
target_compile_features(unison-receiver PRIVATE cxx_std_17)
target_compile_options(unison-receiver PRIVATE /utf-8 /W4)
target_compile_definitions(unison-receiver PRIVATE UNICODE _UNICODE)
target_link_libraries(unison-receiver PRIVATE unison-core ole32 uuid avrt)
